KORAK 7 :


 START snimanje (šablon — MediaRecorder)(UVIJEK ISTI SABLON):



private void startRecording() {
    if (!hasAudioPermission()) {
        audioPermissionLauncher.launch(Manifest.permission.RECORD_AUDIO);
        return;
    }

    try {
        releaseRecorder();

        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.S) {
            mediaRecorder = new MediaRecorder(this);
        } else {
            mediaRecorder = new MediaRecorder();
        }

        mediaRecorder.setAudioSource(MediaRecorder.AudioSource.MIC);
        mediaRecorder.setOutputFormat(MediaRecorder.OutputFormat.MPEG_4);
        mediaRecorder.setAudioEncoder(MediaRecorder.AudioEncoder.AAC);
        mediaRecorder.setOutputFile(outputFile);

        mediaRecorder.prepare();
        mediaRecorder.start();
        isRecording = true;

        btnStartSnimanje.setEnabled(false);
        btnStopSnimanje.setEnabled(true);
        btnPlaySnimak.setEnabled(false);

        Toast.makeText(this, "Snimanje počelo", Toast.LENGTH_SHORT).show();

    } catch (Exception e) {
        isRecording = false;
        Toast.makeText(this, "Greška snimanje", Toast.LENGTH_SHORT).show();
    }
}



KORAK 8 :


 STOP snimanje (šablon):


private void stopRecording() {
    if (!isRecording || mediaRecorder == null) return;

    try {
        mediaRecorder.stop();
    } catch (RuntimeException e) {
        Toast.makeText(this, "Snimak pokvaren", Toast.LENGTH_SHORT).show();
    } finally {
        releaseRecorder();
        isRecording = false;
    }

    btnStartSnimanje.setEnabled(true);
    btnStopSnimanje.setEnabled(false);
    btnPlaySnimak.setEnabled(true);

    Toast.makeText(this, "Snimak sačuvan", Toast.LENGTH_SHORT).show();
}

private void releaseRecorder() {
    if (mediaRecorder != null) {
        mediaRecorder.reset();
        mediaRecorder.release();
        mediaRecorder = null;
    }
}



KORAK 9:



PLAY snimak (šablon — MediaPlayer):



private void playRecording() {
    try {
        stopPlayer();

        mediaPlayer = new MediaPlayer();
        mediaPlayer.setDataSource(outputFile);
        mediaPlayer.prepare();
        mediaPlayer.start();

        Toast.makeText(this, "Pušta se snimak", Toast.LENGTH_SHORT).show();

        mediaPlayer.setOnCompletionListener(mp -> stopPlayer());

    } catch (IOException e) {
        Toast.makeText(this, "Greška reprodukcija", Toast.LENGTH_SHORT).show();
    }
}

private void stopPlayer() {
    if (mediaPlayer != null) {
        if (mediaPlayer.isPlaying()) {
            mediaPlayer.stop();
        }
        mediaPlayer.release();
        mediaPlayer = null;
    }
}

public class MainActivity extends AppCompatActivity {

        outputFile = getExternalCacheDir().getAbsolutePath() + "/audio_record.mp4";

        // KORAK 5b — findViewById

    // KORAK 6 — dozvola
    private void requestAudioPermission() {
        if (ContextCompat.checkSelfPermission(this, Manifest.permission.RECORD_AUDIO)
                == PackageManager.PERMISSION_GRANTED) {
            // OK
        } else {
            audioPermissionLauncher.launch(Manifest.permission.RECORD_AUDIO);
        }
    }

    private boolean hasAudioPermission() {
        return ContextCompat.checkSelfPermission(this, Manifest.permission.RECORD_AUDIO)
                == PackageManager.PERMISSION_GRANTED;
    }

    // KORAK 7 — START (šablon — ne mijenjaš)
    private void startRecording() {
        if (!hasAudioPermission()) {
            audioPermissionLauncher.launch(Manifest.permission.RECORD_AUDIO);
            return;
        }
        try {
            releaseRecorder();
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.S) {
                mediaRecorder = new MediaRecorder(this);
            } else {
                mediaRecorder = new MediaRecorder();
            }
            mediaRecorder.setAudioSource(MediaRecorder.AudioSource.MIC);
            mediaRecorder.setOutputFormat(MediaRecorder.OutputFormat.MPEG_4);
            mediaRecorder.setAudioEncoder(MediaRecorder.AudioEncoder.AAC);
            mediaRecorder.setOutputFile(outputFile);
            mediaRecorder.prepare();
            mediaRecorder.start();
            isRecording = true;

            btnStartSnimanje.setEnabled(false);
            btnStopSnimanje.setEnabled(true);
            btnPlaySnimak.setEnabled(false);
        } catch (Exception e) {
            isRecording = false;
            Toast.makeText(this, "Greška snimanje", Toast.LENGTH_SHORT).show();
        }
    }

    // KORAK 8 — STOP (šablon)
    private void stopRecording() {
        if (!isRecording || mediaRecorder == null) return;
        try {
            mediaRecorder.stop();
        } catch (RuntimeException e) {
            Toast.makeText(this, "Snimak pokvaren", Toast.LENGTH_SHORT).show();
        } finally {
            releaseRecorder();
            isRecording = false;
        }
        btnStartSnimanje.setEnabled(true);
        btnStopSnimanje.setEnabled(false);
        btnPlaySnimak.setEnabled(true);
    }

    private void releaseRecorder() {
        if (mediaRecorder != null) {
            mediaRecorder.reset();
            mediaRecorder.release();
            mediaRecorder = null;
        }
    }

    // KORAK 9 — PLAY (šablon)
    private void playRecording() {
        try {
            stopPlayer();
            mediaPlayer = new MediaPlayer();
            mediaPlayer.setDataSource(outputFile);
            mediaPlayer.prepare();
            mediaPlayer.start();
            mediaPlayer.setOnCompletionListener(mp -> stopPlayer());
        } catch (IOException e) {
            Toast.makeText(this, "Greška reprodukcija", Toast.LENGTH_SHORT).show();
        }
    }

    private void stopPlayer() {
        if (mediaPlayer != null) {
            if (mediaPlayer.isPlaying()) mediaPlayer.stop();
            mediaPlayer.release();
            mediaPlayer = null;
        }
    }

    // KORAK 10 — RESET (opciono)
    private void resetRecording() {
        stopPlayer();
        releaseRecorder();
        isRecording = false;
        btnStartSnimanje.setEnabled(true);
        btnStopSnimanje.setEnabled(false);
        btnPlaySnimak.setEnabled(false);
    }

    // KORAK 11 — čišćenje
    @Override
    protected void onPause() {
        super.onPause();
        stopPlayer();
        releaseRecorder();
        isRecording = false;
    }
}
